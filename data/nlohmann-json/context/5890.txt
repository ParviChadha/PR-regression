    SECTION("duplicate keys: lookups find the first member, iteration all")
    {
        const json_document d = json_document::parse(R"({"a":1,"b":2,"a":3})");
        const json_view v = d.root();
        CHECK(v.size() == 3);
        CHECK(v["a"].materialize() == 1);
        CHECK(v.at("a").materialize() == 1);
        CHECK(v.find("a") == v.begin());
        CHECK(v.find("a").value().materialize() == 1);
        CHECK(v.find("b") == std::next(v.begin()));
        CHECK(v.count("a") == 2); // every member, unlike materialize()
        CHECK(v.count("b") == 1);
        CHECK(v.contains("a"));
        CHECK(v.value("a", 0) == 1);
        CHECK(v.materialize()["a"] == 3); // materialize() keeps the last value, unlike the lookups
        // JSON pointers resolve to the first member at every level
        const json_document dp = json_document::parse(R"({"a":{"b":1},"a":{"b":2,"b":3}})");
        CHECK(dp.root()[json::json_pointer("/a/b")].materialize() == 1);
        CHECK(dp.root().at(json::json_pointer("/a/b")).materialize() == 1);
        CHECK(dp.root().value(json::json_pointer("/a/b"), 0) == 1);
        CHECK(dp.root().materialize()[json::json_pointer("/a/b")] == 3);
        // get<map> keeps the last value, as materialize()
        CHECK((v.get<std::map<std::string, int>>() == std::map<std::string, int> {{"a", 3}, {"b", 2}}));
        // keys of every length class (the 16-byte short compare and memcmp)
        for (const std::size_t n :
                {
                    0u, 1u, 3u, 7u, 8u, 15u, 16u, 17u, 40u
                })
        {
            const std::string key(n, 'k');
            const json_document dk = json_document::parse("{\"" + key + "\":1,\"" + key + "x\":2,\"" + key + "\":3,\"" + key + "\":4}"); // NOLINT(performance-inefficient-string-concatenation)
            CAPTURE(n)
            CHECK(dk.root()[key].materialize() == 1);
            CHECK(dk.root().at(key).materialize() == 1);
            CHECK(dk.root().find(key) == dk.root().begin());
            CHECK(dk.root().value(key, 0) == 1);
            CHECK(dk.root()[key + "x"].materialize() == 2);
        }
        std::string order;
        for (auto it = v.begin(); it != v.end(); ++it)
        {
            order += std::string(it.key().data(), it.key().size()) + it->materialize().dump();
        }
        CHECK(order == "a1b2a3");
        CHECK(v.back().materialize() == 3);
        CHECK(v.materialize() == json::parse(R"({"a":1,"b":2,"a":3})")); // the last value, as parse()
    }

// ---- tests/src/unit-json_view_edit.cpp:727-771  (uses: a, at, b, back, contains, dump, find, materialize, push_back, set, value)
TEST_CASE("json_view edits: an editable document before and after its first edit")
{
    // until the first edit, an editable document is read like a read-only one; views made before the edit
    // keep working after it
    const std::string text = R"({"a":[1,2,{"x":"y"},4],"b":{"c":true,"d":null},"e":"text","f":[]})";
    const json_document read_only = json_document::parse(text);
    json_editable_document d = json_editable_document::parse(text);
    const auto root = d.root();
    const auto same_as_read_only = [&]
    {
        CHECK(root.dump() == read_only.root().dump());
        CHECK(root.dump(2) == read_only.root().dump(2));
        CHECK(root.materialize() == read_only.root().materialize());
        CHECK(root["a"][2]["x"].get_string() == "y");
        CHECK(root["a"].at(3).get<int>() == 4);
        CHECK(root["a"].back().get<int>() == 4);
        CHECK(root.contains("b"));
        CHECK(!root.contains("zz"));
        CHECK(root.find("e")->get_string() == "text");
        CHECK(root.find("zz") == root.end());
        CHECK(root["b"]["c"].get<bool>());
        CHECK(root.value("e", "none") == "text");
        std::size_t count = 0;
        for (const auto m : root["a"])
        {
            CHECK(m.materialize() == read_only.root()["a"][count].materialize());
            ++count;
        }
        CHECK(count == 4);
    };
    same_as_read_only();
    // an edit that changes a scalar in place, then one that moves an array
    d.set(root["e"], "other");
    CHECK(root["e"].get_string() == "other");
    CHECK(root["a"][2]["x"].get_string() == "y");
    CHECK(root["a"].back().get<int>() == 4);
    d.push_back(root["f"], 7);
    CHECK(root["f"][0].get<int>() == 7);
    CHECK(root["f"].back().get<int>() == 7);
    CHECK(root.dump() == R"({"a":[1,2,{"x":"y"},4],"b":{"c":true,"d":null},"e":"other","f":[7]})");
    d.set(root["a"], 2, json::array({1, 2}));
    CHECK(root["a"][2][1].get<int>() == 2);
    CHECK(root["a"].dump() == "[1,2,[1,2],4]");
    CHECK(root.materialize() == json::parse(R"({"a":[1,2,[1,2],4],"b":{"c":true,"d":null},"e":"other","f":[7]})"));
}
