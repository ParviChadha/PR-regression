template <class S, class F>
void test_not_bracketed(const S& solver, const std::string& name,
                        const F& f, Real guess) {
    Real accuracy[] = { 1.0e-4, 1.0e-6, 1.0e-8 };
    Real expected = 1.0;
    for (Real& i : accuracy) {
        Real root = solver.solve(f, i, guess, 0.1);
        if (std::fabs(root - expected) > i) {
            BOOST_FAIL(name << " solver (not bracketed):\n"
                       << "    expected:   " << expected << "\n"
                       << "    calculated: " << root << "\n"
                       << "    accuracy:   " << i);
        }
    }
}

Real hestonPxBoundary(Time maturity, Real eps,
                      const ext::shared_ptr<HestonModel>& model) {

    const AnalyticPDFHestonEngine pdfEngine(model);
    const Real sInit = model->process()->s0()->value();
    const Real xMin = Brent().solve(
            [&](Real x) -> Real { return pdfEngine.cdf(x, maturity) - eps; },
            sInit*1e-3, sInit, sInit*0.001, 1000*sInit);

    return xMin;
}

BOOST_AUTO_TEST_CASE(testAlphaFinderExactBisectionRoot) {
    BOOST_TEST_MESSAGE("Testing AlphaFinder exact bisection roots...");
    for (bool increasing : {false, true}) {
        BOOST_TEST_CONTEXT("increasing=" << increasing) {
            auto form = ext::make_shared<BoundedLinearAlphaForm>(increasing ? 0.0 : 1.0,

            AlphaFinder finder(form);
            std::vector<Volatility> calibratedVols(3);
            Real alpha, scale, lastScale;

                alpha, scale, lastScale, calibratedVols));
            BOOST_REQUIRE(success);
            QL_CHECK_SMALL(alpha - 0.5, 1.0e-12);
            QL_CHECK_SMALL(calibratedVols[0] - 0.8, 1.0e-12);
            QL_CHECK_SMALL(calibratedVols[1] - 0.4, 1.0e-12);
            QL_CHECK_SMALL(calibratedVols[2] - std::sqrt(2.2), 1.0e-12);
        }
