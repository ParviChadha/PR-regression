// ---- test/api/capi/v2/test_capi_v2_result.cpp:1153-1167  (uses: GetContext, context)
TEST_CASE("V2: a cancel requested while a statement is submitted cancels it", "[capi_v2][query_result]") {
	EnvFixture fx;
	auto &con = *reinterpret_cast<duckdb::Connection *>(fx.conn);
	con.context->GetContext().GetRegisteredState().Insert("cancel_while_planning",
	                                                      duckdb::make_shared_ptr<CancelWhilePlanning>(fx.conn));
	duckdb_v2_result_handle result = nullptr;
	REQUIRE(Query(fx.conn, "SELECT 42", &result) == DUCKDB_V2_ERROR_NONE);
	con.context->GetContext().GetRegisteredState().Remove("cancel_while_planning");

	duckdb_v2_data_chunk_handle chunk = nullptr;
	auto status = DUCKDB_V2_RESULT_STEP_STATUS_WAITING;
	REQUIRE(duckdb_v2_result_step(result, &chunk, &status, nullptr) == DUCKDB_V2_ERROR_NONE);
	REQUIRE(status == DUCKDB_V2_RESULT_STEP_STATUS_CANCELLED);
	duckdb_v2_result_destroy(&result);
}

// ---- test/api/test_expanding_statements.cpp:1109-1133  (uses: GetContext, context)
TEST_CASE("Registered states see a StatementBegin and StatementEnd per statement", "[api][expanding_statements]") {
	DuckDB db(nullptr);
	Connection con(db);
	REQUIRE_NO_FAIL(con.Query(SYNTACTIC_ERRORS_POLICY));
	CreateSales(con);
	CreateNumbers(con, 10);
	auto recorder =
	    con.context->GetContext().GetRegisteredState().GetOrCreate<StatementRecorder>("expanding_statements_recorder");

	SECTION("a single statement") {
		REQUIRE_NO_FAIL(con.Query("SELECT 42"));
		REQUIRE(recorder->events ==
		        vector<string> {"QueryBegin", "StatementBegin last", "StatementEnd last", "QueryEnd"});
	}
	SECTION("a statement that expands") {
		REQUIRE_NO_FAIL(con.Query(SALES_PIVOT));
		REQUIRE(recorder->events == vector<string> {"QueryBegin", "StatementBegin", "StatementEnd",
		                                            "StatementBegin last", "StatementEnd last", "QueryEnd"});
	}
	SECTION("a statement that expands and fails before its last statement") {
		REQUIRE_FAIL(con.Query(FAILING_DEFAULT));
		REQUIRE(recorder->events == vector<string> {"QueryBegin", "StatementBegin", "StatementEnd", "StatementBegin",
		                                            "StatementEnd error", "QueryEnd error"});
	}
}
